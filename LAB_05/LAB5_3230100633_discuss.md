import cv2
import numpy as np

def mark_lesion(image, lesion_mask, color=(255, 0, 0)):
    """
    在原图上标记病灶区域
    
    参数:
        image: 原始灰度图像，shape=(H, W)，dtype任意
        lesion_mask: 病灶二值掩膜，0-background, 255-lesion
        color: 标记颜色，(R, G, B)
    
    返回:
        result_img: 三通道彩色图像，病灶区域被标记为指定颜色
    """
    # 1. 归一化原图到 0-255 并转换为 uint8
    gray_uint8 = cv2.normalize(image, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)
    
    # 2. 转换为彩色
    result_img = cv2.cvtColor(gray_uint8, cv2.COLOR_GRAY2RGB)
    
    # 3. 标记病灶区域
    result_img[lesion_mask > 0] = color
    
    return result_img

以上代码会发现背景（intensity = 0）数量太多了，很容易将迭代结果带偏，现在我希望计算一个 image 的有效区域，对 image > 0 的部分进行迭代。具体处理方法为：
1. 添加一个图层 mask，背景为0，信号为1
2. 对信号中的像素做迭代的 for 循环处理


# A:

明白了，你希望 只对有效信号区域（image > 0）计算迭代阈值，而不是把大量背景 0 值算进去，这样迭代结果就不会被背景拉低。

可以修改你的函数如下：

import numpy as np
import cv2

def thresholdIterativeMasked(image, T_init: int = 100, tolerance: int = 1, max_iterations: int = 100, dtype: np.dtype = np.uint8):
    """
    迭代阈值分割，只对 image>0 的有效区域计算
    """
    # 1. 归一化原图
    image = cv2.normalize(image, None, 0, 255, cv2.NORM_MINMAX).astype(dtype)
    
    # 2. 创建有效信号 mask (signal=1, background=0)
    signal_mask = (image > 0)
    
    # 3. 初始化阈值
    T = T_init
    
    for i in range(max_iterations):
        # 仅考虑 signal_mask 内的像素
        signal_pixels = image[signal_mask]
        
        lower = signal_pixels[signal_pixels <= T]
        upper = signal_pixels[signal_pixels > T]
        
        if len(lower) == 0 or len(upper) == 0:
            break
        
        T_new = (lower.mean() + upper.mean()) / 2
        
        if abs(T - T_new) < tolerance:
            break
        T = T_new
    
    # 生成二值 mask
    mask = np.zeros_like(image, dtype=dtype)
    mask[image > T] = 255
    
    return mask, T
