import cv2
import numpy as np

def mark_lesion(image, lesion_mask, color=(255, 0, 0)):
    """
    在原图上标记病灶区域
    
    参数:
        image: 原始灰度图像，shape=(H, W)，dtype任意
        lesion_mask: 病灶二值掩膜，0-background, 255-lesion
        color: 标记颜色，(R, G, B)
    
    返回:
        result_img: 三通道彩色图像，病灶区域被标记为指定颜色
    """
    # 1. 归一化原图到 0-255 并转换为 uint8
    gray_uint8 = cv2.normalize(image, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)
    
    # 2. 转换为彩色
    result_img = cv2.cvtColor(gray_uint8, cv2.COLOR_GRAY2RGB)
    
    # 3. 标记病灶区域
    result_img[lesion_mask > 0] = color
    
    return result_img

import numpy as np
import cv2

def thresholdIterativeMasked(image, T_init: int = 100, tolerance: int = 1, max_iterations: int = 100, dtype: np.dtype = np.uint8):
    """
    迭代阈值分割，只对 image>0 的有效区域计算
    """
    # 1. 归一化原图
    image = cv2.normalize(image, None, 0, 255, cv2.NORM_MINMAX).astype(dtype)
    
    # 2. 创建有效信号 mask (signal=1, background=0)
    signal_mask = (image > 0)
    
    # 3. 初始化阈值
    T = T_init
    
    for i in range(max_iterations):
        # 仅考虑 signal_mask 内的像素
        signal_pixels = image[signal_mask]
        
        lower = signal_pixels[signal_pixels <= T]
        upper = signal_pixels[signal_pixels > T]
        
        if len(lower) == 0 or len(upper) == 0:
            break
        
        T_new = (lower.mean() + upper.mean()) / 2
        
        if abs(T - T_new) < tolerance:
            break
        T = T_new
    
    # 生成二值 mask
    mask = np.zeros_like(image, dtype=dtype)
    mask[image > T] = 255
    
    return mask, T

# A:

明白了，你希望 **空间权重不再是单独给 x/y 加权，而是根据像素到图像中心的距离来加权**。距离越近中心，权重越高。这里可以用 **中心距离反比或高斯权重** 实现。最简单的办法是使用归一化的欧氏距离，并乘以 `spatial_weight` 加到特征上。下面是修改后的 `improved_kmeanCluster` 核心部分：

```python id="8x9c1q"
def improved_kmeanCluster(image, K_list=[2,3,4], spatial_weights=[0,0.05,0.1,0.2], dtype=np.uint8):
    """
    Improved K-means clustering with spatial weighting based on distance to image center.
    """
    
    img = normalize(image, dtype=dtype)
    h, w = img.shape
    cx, cy = w / 2, h / 2  # 图像中心
    
    # 生成坐标网格
    X, Y = np.meshgrid(np.arange(w), np.arange(h))
    # 计算每个像素到中心的距离
    dist_to_center = np.sqrt((X - cx)**2 + (Y - cy)**2)
    # 归一化距离: 越近中心值越大
    spatial_map = 1 - dist_to_center / dist_to_center.max()
    
    best_mask = None
    best_labels = None
    best_score = -np.inf
    best_params = {}
    all_results = []

    for K in K_list:
        for spatial_weight in spatial_weights:
            features = [img.flatten()]
            if spatial_weight > 0:
                features.append((spatial_map.flatten() * spatial_weight))
            features_stack = np.stack(features, axis=1)
            
            kmeans = KMeans(n_clusters=K, n_init=10, random_state=0)
            kmeans.fit(features_stack)
            labels = kmeans.labels_.reshape(h, w)
            
            cluster_means = [img[labels==i].mean() for i in range(K)]
            lesion_label = np.argmax(cluster_means)
            mask = np.zeros_like(img, dtype=dtype)
            mask[labels==lesion_label] = 255
            
            # 病灶平均亮度作为评分
            score = img[mask==255].mean()
            
            if score > best_score:
                best_score = score
                best_mask = mask
                best_labels = labels
                best_params = {'K': K, 'spatial_weight': spatial_weight, 'score': score}

            all_results.append({
                "labels": labels,
                "mask": mask,
                "params": {'K': K, 'spatial_weight': spatial_weight, 'score': score}
            })
    
    return best_labels, best_mask, best_params, all_results
```

✅ **修改点说明：**

1. 增加了 `cx, cy` 作为图像中心点。
2. 用 `dist_to_center = sqrt((X-cx)^2 + (Y-cy)^2)` 计算每个像素到中心的欧氏距离。
3. 生成空间权重图 `spatial_map = 1 - dist/max(dist)`，使越靠近中心值越大（归一化到 [0,1]）。
4. 在特征向量中加入 `spatial_map * spatial_weight` 作为额外特征，而不再单独加 x/y 坐标。

这样 K-means 会更倾向于把中心附近亮度高的区域聚成同一簇，从而改善病灶分割效果。
