import numpy as np

def binary_dilation_basic(binary_img, kernel=None):
    img = (binary_img > 0).astype(np.uint8)

    if kernel is None:
        kernel = np.ones((3, 3), dtype=np.uint8)

    kh, kw = kernel.shape
    pad_h, pad_w = kh // 2, kw // 2

    padded = np.pad(img, ((pad_h, pad_h), (pad_w, pad_w)),
                    mode='constant', constant_values=0)
    out = np.zeros_like(img, dtype=np.uint8)

    kernel_ones = (kernel == 1)

    for i in range(img.shape[0]):
        for j in range(img.shape[1]):
            region = padded[i:i+kh, j:j+kw]
            if np.any(region[kernel_ones] == 1):
                out[i, j] = 1

    return out.astype(bool)
2）按公式实现 hole fill
def hole_fill(binary_img, kernel=None, return_debug=False):
    """
    对二值图像做 hole filling

    参数
    ----
    binary_img : 2D array
        前景为1/True，背景为0/False
    kernel : 2D array
        结构元素，默认 3x3 全1
    return_debug : bool
        是否返回中间结果

    返回
    ----
    filled : bool array
        填孔后的结果
    debug : dict
        可选，中间变量
    """
    A = (binary_img > 0).astype(bool)

    if kernel is None:
        kernel = np.ones((3, 3), dtype=np.uint8)

    # A^c
    Ac = ~A

    # X0: 从图像边界上的背景开始
    X = np.zeros_like(A, dtype=bool)
    X[0, :] = Ac[0, :]
    X[-1, :] = Ac[-1, :]
    X[:, 0] = X[:, 0] | Ac[:, 0]
    X[:, -1] = X[:, -1] | Ac[:, -1]

    # 迭代: X_k = (X_(k-1) ⊕ B) ∩ A^c
    while True:
        X_next = binary_dilation_basic(X, kernel=kernel) & Ac
        if np.array_equal(X_next, X):
            break
        X = X_next

    # X 为与边界连通的背景
    # hole = A^c - X
    holes = Ac & (~X)

    # phi(A) = A ∪ hole
    filled = A | holes

    if return_debug:
        debug = {
            "A": A,
            "Ac": Ac,
            "Xk": X,
            "holes": holes
        }
        return filled, debug

    return filled

# Answer

根据您的需求，我编写了一个函数，将二值掩膜以半透明红色的形式叠加在背景图像上：

python
import cv2
import numpy as np

def overlay_mask(img_background, img_mask, alpha=0.5, color=(0, 0, 255)):
    """
    将二值掩膜以半透明颜色叠加在背景图像上
    
    参数:
        img_background: 背景图像 (灰度图或彩色图)
        img_mask: 二值掩膜图像 (非0区域将被叠加颜色)
        alpha: 透明度系数 (0~1, 0完全透明, 1完全不透明)
        color: 叠加颜色 BGR格式, 例如红色 (0, 0, 255)
    
    返回:
        叠加后的彩色图像 (BGR格式)
    """
    # 确保背景图像是彩色图 (3通道)
    if len(img_background.shape) == 2:
        img_bg_color = cv2.cvtColor(img_background, cv2.COLOR_GRAY2BGR)
    else:
        img_bg_color = img_background.copy()
    
    # 确保掩膜是二值图 (0或255)
    if img_mask.dtype != np.uint8:
        img_mask = img_mask.astype(np.uint8)
    
    # 将掩膜二值化：非0区域变为255
    _, binary_mask = cv2.threshold(img_mask, 0, 255, cv2.THRESH_BINARY)
    
    # 创建纯色掩膜层 (与背景相同尺寸，纯色)
    overlay = np.full_like(img_bg_color, color, dtype=np.uint8)
    
    # 创建掩膜区域 (3通道用于索引)
    mask_3ch = binary_mask > 0
    
    # 创建结果图像
    result = img_bg_color.copy()
    
    # 只对掩膜有效的区域进行叠加
    result[mask_3ch] = cv2.addWeighted(
        img_bg_color[mask_3ch], 
        1 - alpha, 
        overlay[mask_3ch], 
        alpha, 
        0
    )
    
    return result

def overlay_mask_simple(img_background, img_mask, alpha=0.5):
    """简化版本，固定红色半透明"""
    # 确保彩色图
    if len(img_background.shape) == 2:
        img_bg = cv2.cvtColor(img_background, cv2.COLOR_GRAY2BGR)
    else:
        img_bg = img_background.copy()
    
    # 创建红色层
    red_layer = np.zeros_like(img_bg)
    red_layer[:, :, 2] = 255  # BGR中红色通道是第2个索引
    
    # 创建掩膜（非0区域）
    mask = img_mask > 0
    
    # 混合
    result = img_bg.copy()
    result[mask] = (1 - alpha) * img_bg[mask] + alpha * red_layer[mask]
    
    return result.astype(np.uint8)

# Question

除了 mse 和 psnr 以外，还有什么可供参考的指标来评估图像修复程度吗？


# Answer

除了 MSE (均方误差) 和 PSNR (峰值信噪比)，还有一些其他常用的评估图像修复质量的指标，它们能从不同的角度评估恢复图像与原图的差异，特别是在细节、结构和感知质量方面。下面列出一些常见的指标：

1. SSIM (结构相似性指数)
定义：SSIM 衡量两幅图像在亮度、对比度和结构方面的相似性。它通常比 MSE 和 PSNR 更接近人类感知，因为它考虑了结构信息。
公式：SSIM 使用亮度、对比度和结构对比三者的结合来计算相似性，值域为 [0, 1]，越接近 1 表示越相似。

应用：SSIM 在图像恢复（如去噪、去模糊、图像压缩等）中非常有用，能够更好地反映图像的视觉质量。

示例代码：

from skimage.metrics import structural_similarity as ssim
ssim_value = ssim(original_image, restored_image)
2. NIQE (自然图像质量评估)
定义：NIQE 是一个无参考图像质量评估指标，专门设计用于衡量图像质量的自然性，它不需要原始参考图像。NIQE 基于自然图像的统计特性来评估图像的质量。

应用：通常用于图像压缩、去噪等处理后的质量评价，尤其是在没有原始图像时非常有用。

示例代码：

from skimage.metrics import niqe
niqe_value = niqe(restored_image)
3. FSIM (特征相似性指数)
定义：FSIM 通过衡量图像在亮度、颜色和结构上的相似性来评估图像质量，特别注重图像的结构和纹理信息。

应用：与 SSIM 类似，FSIM 也能评估结构相似性，适用于图像压缩、去噪、去模糊等任务中。

示例代码：

from skimage.metrics import structural_similarity as ssim
# 计算结构相似度 (SSIM) 和 FSIM 等其他指标
4. VIF (视觉信息保真度)
定义：VIF 衡量图像在视觉系统中的信息保真度。它基于人眼视觉系统的工作原理来设计，能够更准确地反映图像的感知质量。
应用：VIF 常用于高质量图像处理任务的评估，能够提供更符合视觉系统感知的质量评分。
5. MSSSIM (多尺度结构相似性)
定义：MSSSIM 是对 SSIM 的扩展，使用多尺度的方式计算图像的结构相似性，能够更加细致地评估图像不同尺度下的质量。
应用：常用于图像去噪、图像压缩等多尺度图像质量评估任务。
6. RMS (均方根误差)
定义：RMS 是 MSE 的平方根，作为误差度量，它可以提供一种衡量图像像素级差异的方式。

公式：RMS = √(Σ|I1 - I2|² / N)，其中 I1 和 I2 分别是原图和恢复图的像素，N 是像素总数。

示例代码：

import numpy as np
rms_value = np.sqrt(np.mean((original_image - restored_image) ** 2))
7. LPIPS (学习感知图像相似性)
定义：LPIPS 是一种基于深度学习的方法，通过学习图像的感知差异来评估图像的质量。它在许多现代图像修复任务（如图像超分辨率、生成对抗网络）中有很好的表现。
应用：LPIPS 更注重感知质量，能更好地反映人眼感知的图像质量。
8. 相对误差 (Relative Error)
定义：该指标计算恢复图像与原图之间的绝对误差和原图的相对比率。

公式：RE = ||(I - I')|| / ||I||，其中 I 是原始图像，I' 是恢复图像。

示例代码：

relative_error = np.linalg.norm(original_image - restored_image) / np.linalg.norm(original_image)
总结
SSIM 和 PSNR 是传统图像质量评估中最常见的指标，主要用于评估图像的结构和亮度信息。
NIQE、FSIM 和 VIF 等则是更现代的评估方法，尤其是在没有参考图像时，它们能更好地反映人类视觉系统的感知质量。
RMS 和 MSE 都是基于像素的误差度量，但它们并不总是能很好地捕捉图像的视觉感知质量。

你可以根据你的项目需求选择合适的评估指标，结合 PSNR/SSIM 来综合评估图像修复质量。如果你有原图并希望精确评估视觉质量，SSIM 和 PSNR 是最常用的指标。如果没有参考图像，考虑使用 NIQE 或 LPIPS 等无参考指标。

我现在希望用 NLM 来实现对 Rician Noise 的去除，请问有没有现成的 NLM 函数可供调用？如果有的话能否告诉我如何使用，必须的参数有哪些？

# Answer

是的，有现成的 NLM（Non‑Local Means, 非局部均值） 函数可以直接调用，用于图像去噪，包括处理 Rician 噪声的效果通常也不错（虽然严格来讲 Rician noise 的统计特性不同于高斯噪声，但 NLM 对结构保留能力强，实际效果可观）。

在 Python 里，常用的现成实现是 scikit‑image 提供的：

🔹 skimage.restoration.denoise_nl_means

它支持灰度图像、多通道图像、不同大小的搜索窗口和补丁。下面是怎么调用和参数说明。

📌 1) 基本用法
from skimage.restoration import denoise_nl_means, estimate_sigma
a) 对一幅噪声图像进行 NLM 去噪
denoised = denoise_nl_means(
    noisy_image,
    h=1.0,
    patch_size=7,
    patch_distance=11,
    fast_mode=True,
    multichannel=False
)
🧠 2) 重要参数解释
参数	说明
image	输入噪声图，float 图像（通常归一化到 0,1）
h	平滑参数：权重衰减阈值，值越大去噪越强，但也更容易过平滑
patch_size	补丁大小（局部小块），常用 5~7
patch_distance	搜索相似 patch 的最大距离（即“非局部”范围）
fast_mode	是否启用快速近似算法，True 更快但稍微牺牲精度
multichannel	是否是多通道图（如 RGB），灰度图设为 False
🧪 3) 估计噪声强度（自动设定 h）

可以先估计 noise sigma：

sigma_est = np.mean(estimate_sigma(noisy_image, multichannel=False))

然后设定 h：

denoised = denoise_nl_means(
    noisy_image,
    h=1.15 * sigma_est,
    patch_size=5,
    patch_distance=6,
    fast_mode=True
)

一般建议：

h ≈ 0.8 ~ 1.3 * sigma
📌 4) Rician Noise 的注意点

Rician noise 与高斯噪声不同，它在低信号区域偏差更大。
虽然 denoise_nl_means 默认针对高斯噪声设计，但真实使用中它仍然能对 Rician 去噪产生明显效果（尤其在 MRI 上）。

若你希望更严谨地处理 Rician 噪声，还可以：

方法 A：先做 RiBo sta 平移校正（bias correction）

对 MRI Rician noise，一个常见做法是先做：

noisy = sqrt( noisy_raw^2 - 2 * sigma^2 )

再做 NLM 去噪。

估计 σ 可以用：

sigma_est = np.mean(estimate_sigma(noisy_raw, multichannel=False))

注意：Rician bias 校正并不是必须，但对高噪声场景有帮助。

方法 B：对平方图像做 NLM

有些论文建议对 squared magnitude 做去噪：

denoised2 = denoise_nl_means(noisy_image**2, ...)
denoised = sqrt(denoised2)
